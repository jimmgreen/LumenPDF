# LUMEN 界面库：优先使用 git submodule（third_party/LUMENUI），否则回退到并列目录 ../LUMENUI（本地开发）。
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/LUMENUI/CMakeLists.txt")
    set(LUMEN_DEFAULT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/LUMENUI")
else()
    set(LUMEN_DEFAULT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../LUMENUI")
endif()

if(NOT EXISTS "${MUPDF_LIBRARY_DIR}/libmupdf.lib")
    message(FATAL_ERROR "Run scripts/bootstrap.ps1 to build MuPDF first.")
endif()

add_subdirectory("${LUMEN_SOURCE_DIR}" lumen)
add_library(pdf_core STATIC core/document.cpp core/conversion.cpp core/platform.cpp core/font_catalog.cpp core/signature.cpp)
target_include_directories(pdf_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(pdf_core SYSTEM PRIVATE "${MUPDF_SOURCE_DIR}/include")
target_compile_definitions(pdf_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(pdf_core PRIVATE /W4 /permissive- /utf-8 /EHsc)
target_link_libraries(pdf_core PUBLIC
    "${MUPDF_LIBRARY_DIR}/libmupdf.lib" "${MUPDF_LIBRARY_DIR}/libthirdparty.lib"
    "${MUPDF_LIBRARY_DIR}/libresources.lib" "${MUPDF_LIBRARY_DIR}/libharfbuzz.lib"
    "${MUPDF_LIBRARY_DIR}/libpkcs7.lib"
    crypt32 bcrypt ws2_32 ole32 oleaut32 shlwapi shell32 comdlg32 uuid dwrite gdiplus)
lumen_add_executable(LumenPDF app/main.cpp app/canvas.cpp app/annotation_painter.cpp app/inline_editor.cpp app/text_format_bar.cpp app/application.cpp app/document_tabs.cpp app/annotation_tools.cpp app/signature_tools.cpp app/form_tools.cpp app/redaction_tools.cpp app/print_preview.cpp app/clipboard_image.cpp app/clipboard_tools.cpp app/view_rotation.cpp app/zoom_tools.cpp app/speech.cpp app/read_aloud.cpp app/merge_tools.cpp app/recent_files.cpp app/recent_view.cpp app/print_job.cpp app/shell_integration.cpp app/app_log.cpp app/app_settings.cpp app/update_core.cpp app/update_net.cpp app/update_install.cpp app/update_ui.cpp app/app.rc)
target_link_libraries(LumenPDF PRIVATE pdf_core comctl32 uxtheme gdi32 sapi ole32 winhttp bcrypt)
target_compile_definitions(LumenPDF PRIVATE LPDF_VERSION="${PROJECT_VERSION}" LPDF_VERSION_W=L"${PROJECT_VERSION}")
# 生成链接映射：Release 无 PDB 时把崩溃日志里的“模块+偏移”对应到函数。
